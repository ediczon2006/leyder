function goTo(page) {
  if (!PAGE_HANDLERS[page]) return;
  document.querySelectorAll('.page').forEach(p => p.classList.remove('active'));
  document.querySelectorAll('.nav-item').forEach(n => n.classList.remove('active'));
  const targetId = ['favoritos','recientes','analizador'].includes(page) ? 'page-explorador' : 'page-' + page;
  const el = document.getElementById(targetId);
  if (el) el.classList.add('active');
  document.querySelectorAll('.nav-item').forEach(n => {
    if (n.dataset.page === page) n.classList.add('active');
  });
  PAGE_HANDLERS[page]();
  updateSidebar();
}

    const [total, activos, cortados, pagosMes] = await Promise.all([
      IDB.count('clientes'),
      IDB.countByIndex('clientes', 'estado', 'ACTIVO'),
      IDB.countByIndex('clientes', 'estado', 'CORTADO'),
      IDB.getByIndex('pagos', 'mes', mes)
    ]);

    const cobrado = (pagosMes || []).reduce((s, p) => s + (Number(p.monto) || 0), 0);

    const kpiEl = document.getElementById('kpiGrid');
    if (kpiEl) {
      kpiEl.innerHTML = `

${(cortados||0).toLocaleString()}

Cobrado en ${esc(mes)}
`;
    }

    const zonaCounts = await Promise.all(ZONAS.map(async z => {
      const [totZ, actZ, cortZ] = await Promise.all([
        IDB.countByIndex('clientes', 'zona', z),
        IDB.countByIndex('clientes', 'zona_estado', [z, 'ACTIVO']),
        IDB.countByIndex('clientes', 'zona_estado', [z, 'CORTADO'])

⚠️ Clientes con mensualidad pendiente en ${esc(mes)}

async function renderClientes(page = 0) {
  _pg.clientes.page = page;
  const zona   = document.getElementById('filtZona')?.value || '';
  const estado = document.getElementById('filtEstado')?.value || '';
  const plan   = document.getElementById('filtPlan')?.value || '';
  const bus    = safe(document.getElementById('filtBus')?.value || '').toLowerCase();

  const tbody = document.getElementById('bodyClientes');
  tbody.innerHTML = 'Cargando...';

  try {
    let indexName = null;
    let range = null;

    if (zona && estado) {
      indexName = 'zona_estado';
      range = IDBKeyRange.only([zona, estado]);
    } else if (zona) {
      indexName = 'zona';
      range = IDBKeyRange.only(zona);
    } else if (estado) {
      indexName = 'estado';
      range = IDBKeyRange.only(estado);
    } else if (plan) {
      indexName = 'plan';
      range = IDBKeyRange.only(Number(plan));
    }

    const filter = c => {
      if (zona && c.zona !== zona) return false;
      if (estado && c.estado !== estado) return false;
      if (plan && Number(c.plan) !== Number(plan)) return false;
      if (bus) {
        const text = `${c.id} ${c.nombre || ''} ${c.ip || ''} ${c.dni || ''} ${c.celular || ''} ${c.direccion || ''}`.toLowerCase();
        if (!text.includes(bus)) return false;
      }
      return true;
    };

    const { results, total } = await IDB.query('clientes', {
      indexName,
      range,
      filter,
      limit: PAGE_SIZE,
      offset: page * PAGE_SIZE
    });

    _pg.clientes.total = total;
    document.getElementById('contadorClientes').textContent =
      `👥 ${total.toLocaleString()} cliente(s) encontrado(s)`;

    if (!results.length) {
      tbody.innerHTML = '
👥
Sin clientes registrados con ese filtro
';
    } else {
      results.forEach(c => _clientesCache.set(Number(c.id), c));

      tbody.innerHTML = results.map(c => `
        
          ${Number(c.id)}
          ${esc(c.nombre)}
          ${esc(c.zona)}
          ${esc(c.ip) || '—'}
          S/. ${Number(c.plan || 0).toFixed(2)}
          ${badgeCable(c.cable)}
          ${badgeEstado(c.estado)}
          ${c.fecha_inicio ? esc(c.fecha_inicio) : '—'}
          ${esc(c.celular) || '—'}
          
            

              
              
              
              
            

          
        `).join('');
    }

    renderPagination('paginacionClientes', 'paginfoClientes', _pg.clientes, renderClientes);
  } catch(err) {
    console.error('renderClientes:', err);
    tbody.innerHTML = 'Error cargando clientes';
  }
}

async function eliminarCliente(id) {
  const c = await IDB.get('clientes', Number(id));
  if (!c) return;
  if (!confirm(`¿Eliminar definitivamente a "${c.nombre}"?\nEsta acción eliminará también su historial de pagos y cortes.`)) return;
  try {
    await IDB.eliminarClienteCompleto(Number(id));
    _clientesCache.delete(Number(id));
    toast('Cliente y su historial eliminados', 'warn');
    renderClientes(_pg.clientes.page);
    updateSidebar();
  } catch(err) {
    console.error('eliminarCliente:', err);
    toast('Error al eliminar cliente', 'error');
  }
}

function _codigoOp(pagoId, fechaStr) {
  const now = new Date();
  let yy = String(now.getFullYear()).slice(2);
  let mm = String(now.getMonth() + 1).padStart(2, '0');
  let dd = String(now.getDate()).padStart(2, '0');

  if (fechaStr) {
    const limpio = String(fechaStr).replace(/,/g, ' ').trim();
    const partes = limpio.match(/(\d{1,2})[\/\-](\d{1,2})[\/\-](\d{2,4})/);
    if (partes) {
      dd = partes[1].padStart(2, '0');
      mm = partes[2].padStart(2, '0');
      yy = partes[3].length === 4 ? partes[3].slice(2) : partes[3];
    }
  }
  return `LYX-${yy}${mm}${dd}-${String(pagoId || 1).padStart(5, '0')}`;
}

function _fila(label, value, opts = {}) {
  const color = opts.highlight ? '#1E40AF' : '#374151';
  const fw    = opts.highlight ? '700' : '500';
  return `

    ${esc(label)}
    ${esc(String(value))}
  
`;
}

async function generarRecibo(pagoId) {
  try {
    const pago = await IDB.get('pagos', Number(pagoId));
    if (!pago) return;
    const cli  = await IDB.get('clientes', Number(pago.id_cliente));

    const planNombre = { 50:'Plan Básico', 60:'Plan Plus', 65:'Plan Normal' }[Number(cli?.plan)] || `Plan S/. ${cli?.plan || 0}`;

    let mesTexto = pago.mes;
    try {
      const [ay, am] = pago.mes.split('-');
      mesTexto = new Date(+ay, +am - 1, 1).toLocaleString('es-PE', { month:'long', year:'numeric' });
    } catch(_) {}

    const codigoOp = _codigoOp(pagoId, pago.fecha);
    _reciboActual = { pago, cli, planNombre, codigoOp, mesTexto };

    const btnWa = document.getElementById('btnWhatsappRecibo');
    if (btnWa) {
      btnWa.textContent = cli?.celular 
        ? `💬 Enviar por WhatsApp (${cli.celular})` 
        : '💬 Enviar por WhatsApp al Cliente';
    }

    document.getElementById('reciboMonto').textContent = `S/. ${Number(pago.monto || 0).toFixed(2)}`;
    document.getElementById('reciboOperacion').textContent = codigoOp;
    document.getElementById('reciboDetalles').innerHTML = [
      _fila('Nombre del cliente',  cli?.nombre || '—'),
      _fila('Zona de servicio',    cli?.zona   || '—'),
      cli?.fecha_inicio ? _fila('Fecha de inicio', cli.fecha_inicio) : '',
      _fila('Plan contratado',     planNombre,       { highlight: true }),
      _fila('Período pagado',      mesTexto,         { highlight: true }),
      _fila('Método de pago',      pago.metodo || 'Efectivo'),
      _fila('Dirección IP',        cli?.ip     || '—'),
      _fila('Fecha de pago',       pago.fecha  || '—'),
      pago.observaciones ? _fila('Nota', pago.observaciones) : '',
    ].filter(Boolean).join('');

    abrirModal('modalRecibo');
  } catch(err) {
    console.error('generarRecibo:', err);
    toast('No se pudo generar el recibo', 'error');
  }
}

async function importarExcel(tipo, fileInput) {
  const file = fileInput.files[0];
  if (!file) return;
  if (file.size > 30 * 1024 * 1024) { toast('Archivo demasiado grande (máx 30 MB)','error'); return; }

  const resEl = document.getElementById('importResult');
  resEl.innerHTML = '
 Procesando archivo Excel...
';

  const reader = new FileReader();
  reader.onload = async e => {
    try {
      const wb   = XLSX.read(e.target.result, { type: 'array' });
      const ws   = wb.Sheets[wb.SheetNames[0]];
      const rows = XLSX.utils.sheet_to_json(ws, { defval: '' });
      let ok = 0, err = 0;

      if (tipo === 'clientes') {
        const items = [];
        for (const r of rows) {
          const get = (...keys) => {
            for (const k of keys) {
              const found = Object.keys(r).find(rk => rk.toLowerCase().replace(/[áéíóú]/g,'a').trim() === k.toLowerCase());
              if (found !== undefined && r[found] !== '') return String(r[found]).trim();
            }
            return '';
          };

          const nombre       = get('nombre','name');
          const ip           = get('ip','dirección ip','direccion ip');
          const zona         = get('zona','zone');
          const plan         = Number(get('plan','plan s/.','precio'));
          const cable        = get('cable').toLowerCase() === 'sí' || get('cable').toLowerCase() === 'si' || get('cable') === '1' || get('cable').toLowerCase() === 'true';
          const fecha_inicio = get('fecha_inicio','fecha inicio','inicio','f. inicio','f.inicio','f_inicio');

          if (!V.nombre(nombre) || !V.ip(ip) || !V.zona(zona) || !V.plan(plan)) {
            err++;
            continue;
          }

          items.push({
            nombre: safe(nombre, 100),
            dni: safe(get('dni'), 8),
            celular: safe(get('celular','telefono'), 15),
            fecha_inicio: safe(fecha_inicio, 15) || fechaHoy(),
            direccion: safe(get('direccion','dirección'), 200),
            ip: safe(ip, 15),
            zona,
            plan,
            cable,
            estado: ESTADOS.includes(get('estado').toUpperCase()) ? get('estado').toUpperCase() : 'ACTIVO',
            fecha_registro: fechaHora(),
            observaciones: safe(get('observaciones'), 400),
          });
          ok++;
        }
        await IDB.bulkPut('clientes', items);
      }

      if (tipo === 'pagos') {
        const clientes = await IDB.getAll('clientes');
        const byNombre = new Map(clientes.map(c => [c.nombre.toLowerCase().trim(), c]));
        const items = [];

        for (const r of rows) {
          const get = (...keys) => {
            for (const k of keys) {
              const found = Object.keys(r).find(rk => rk.toLowerCase().replace(/[áéíóú]/g,'a').trim() === k.toLowerCase());
              if (found !== undefined && r[found] !== '') return String(r[found]).trim();
            }
            return '';
          };

          const nombre_cli = get('nombre_cliente','cliente','nombre');
          const mes        = get('mes','month');
          const monto      = Number(get('monto','monto s/.'));
          const cli        = byNombre.get(nombre_cli.toLowerCase().trim());

          if (!cli || !V.mes(mes) || !V.monto(monto)) {
            err++;
            continue;
          }

          items.push({
            id_cliente: Number(cli.id),
            mes,
            monto: Number(monto.toFixed(2)),
            metodo: safe(get('metodo','método'), 50) || 'Efectivo',
            observaciones: safe(get('observaciones'), 200),
            fecha: fechaHora(),
          });
          ok++;
        }
        await IDB.bulkPut('pagos', items);
      }

      updateSidebar();
      fileInput.value = '';
      resEl.innerHTML = `
✅ Importación completada: ${ok} registros agregados${err ? `, ${err} omitidos por datos inválidos` : ''}.
`;
      toast(`${ok} registros importados correctamente`);
    } catch(ex) {
      console.error('importarExcel:', ex);
      resEl.innerHTML = `
❌ Error procesando el archivo: ${esc(ex.message)}
`;
      toast('Error al importar archivo','error');
    }
  };

async function restaurarBackupCompleto(fileInput) {
  const file = fileInput.files[0];
  if (!file) return;
  const resEl = document.getElementById('backupResult');
  if (resEl) resEl.innerHTML = '
 Validando y restaurando respaldo...
';

  const reader = new FileReader();
  reader.onload = async e => {
    try {
      // Prevención de Prototype Pollution mediante reviver estricto
      const data = JSON.parse(e.target.result, (key, value) => {
        if (key === '__proto__' || key === 'constructor' || key === 'prototype') {
          return undefined;
        }
        return value;
      });

      if (!data || (!Array.isArray(data.clientes) && !Array.isArray(data.pagos))) {
        throw new Error('El archivo no tiene el formato de respaldo válido de Leytrix.');
      }

      const rawClientes = Array.isArray(data.clientes) ? data.clientes : [];
      const rawPagos    = Array.isArray(data.pagos) ? data.pagos : [];
      const rawCortes   = Array.isArray(data.cortes) ? data.cortes : [];

      if (!confirm(`¿Restaurar copia de seguridad?\nSe importarán ${rawClientes.length} clientes y ${rawPagos.length} pagos.\nLos datos se guardarán de forma permanente.`)) {
        if (resEl) resEl.innerHTML = '';
        fileInput.value = '';
        return;
      }

      // Sanitización y validación estricta de cada objeto antes de la inserción
      const sanitizedClientes = rawClientes.map(c => ({
        id: Number(c.id) || undefined,
        nombre: safe(c.nombre, 100),
        dni: safe(c.dni, 8),
        celular: safe(c.celular, 15),
        fecha_inicio: safe(c.fecha_inicio, 15),

        estado: co.estado === 'RECONECTADO' ? 'RECONECTADO' : 'CORTADO'
      })).filter(co => co.id_cliente);

      let [cliOk, pagOk, corOk] = await Promise.all([
        IDB.bulkPut('clientes', sanitizedClientes),
        IDB.bulkPut('pagos', sanitizedPagos),

❌ Error restaurando el respaldo: ${esc(err.message)}

//  EXPORTAR A SCRIPT MYSQL (.sql) BLINDADO CONTRA INYECCIÓN
// ══════════════════════════════════════════════════════
async function exportarMySQL() {
  const btn = document.getElementById('btnExpMySQL');
  if (btn) { btn.disabled = true; btn.textContent = '⏳ Generando script SQL...'; }
  try {
    const [clientes, pagos, cortes] = await Promise.all([
      IDB.getAll('clientes'),
      IDB.getAll('pagos'),
      IDB.getAll('cortes'),
    ]);

    const fecha = new Date().toISOString();
    let sql = `-- ========================================================\n`;
    sql += `-- LEYTRIX ISP - BASE DE DATOS COMPATIBLE CON MYSQL\n`;
    sql += `-- Generado: ${fecha}\n`;
    sql += `-- Compatible con MySQL 5.7+ / MySQL 8.0+ / MariaDB / phpMyAdmin\n`;
    sql += `-- ========================================================\n\n`;
    sql += `CREATE DATABASE IF NOT EXISTS \`leytrix_isp\` DEFAULT CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;\n`;
    sql += `USE \`leytrix_isp\`;\n\n`;

    // Tabla Clientes
    sql += `-- Estructura de tabla para \`clientes\`\n`;
    sql += `CREATE TABLE IF NOT EXISTS \`clientes\` (\n`;
    sql += `  \`id\` int(11) NOT NULL AUTO_INCREMENT,\n`;
    sql += `  \`nombre\` varchar(150) NOT NULL,\n`;
    sql += `  \`dni\` varchar(10) DEFAULT NULL,\n`;
    sql += `  \`celular\` varchar(20) DEFAULT NULL,\n`;
    sql += `  \`fecha_inicio\` varchar(20) DEFAULT NULL,\n`;
    sql += `  \`direccion\` varchar(255) DEFAULT NULL,\n`;
    sql += `  \`ip\` varchar(45) NOT NULL,\n`;
    sql += `  \`zona\` varchar(100) NOT NULL,\n`;
    sql += `  \`plan\` decimal(10,2) NOT NULL,\n`;
    sql += `  \`cable\` tinyint(1) NOT NULL DEFAULT 0,\n`;
    sql += `  \`estado\` varchar(30) NOT NULL DEFAULT 'ACTIVO',\n`;
    sql += `  \`fecha_registro\` varchar(30) DEFAULT NULL,\n`;
    sql += `  \`observaciones\` text DEFAULT NULL,\n`;
    sql += `  PRIMARY KEY (\`id\`),\n`;
    sql += `  KEY \`idx_zona\` (\`zona\`),\n`;
    sql += `  KEY \`idx_estado\` (\`estado\`)\n`;
    sql += `) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;\n\n`;

    if (clientes.length) {
      sql += `INSERT INTO \`clientes\` (\`id\`, \`nombre\`, \`dni\`, \`celular\`, \`fecha_inicio\`, \`direccion\`, \`ip\`, \`zona\`, \`plan\`, \`cable\`, \`estado\`, \`fecha_registro\`, \`observaciones\`) VALUES\n`;
      const vals = clientes.map(c => `(${Number(c.id)}, ${escSql(c.nombre)}, ${escSql(c.dni)}, ${escSql(c.celular)}, ${escSql(c.fecha_inicio)}, ${escSql(c.direccion)}, ${escSql(c.ip)}, ${escSql(c.zona)}, ${Number(c.plan || 0).toFixed(2)}, ${c.cable ? 1 : 0}, ${escSql(c.estado)}, ${escSql(c.fecha_registro)}, ${escSql(c.observaciones)})`);
      sql += vals.join(',\n') + ';\n\n';
    }

    // Tabla Pagos
    sql += `-- Estructura de tabla para \`pagos\`\n`;
    sql += `CREATE TABLE IF NOT EXISTS \`pagos\` (\n`;
    sql += `  \`id\` int(11) NOT NULL AUTO_INCREMENT,\n`;
    sql += `  \`id_cliente\` int(11) NOT NULL,\n`;
    sql += `  \`mes\` varchar(20) NOT NULL,\n`;
    sql += `  \`monto\` decimal(10,2) NOT NULL,\n`;
    sql += `  \`metodo\` varchar(50) DEFAULT 'Efectivo',\n`;
    sql += `  \`fecha\` varchar(30) DEFAULT NULL,\n`;
    sql += `  \`observaciones\` text DEFAULT NULL,\n`;
    sql += `  PRIMARY KEY (\`id\`),\n`;
    sql += `  KEY \`fk_cliente_pago\` (\`id_cliente\`),\n`;
    sql += `  KEY \`idx_mes\` (\`mes\`)\n`;
    sql += `) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;\n\n`;

    if (pagos.length) {
      sql += `INSERT INTO \`pagos\` (\`id\`, \`id_cliente\`, \`mes\`, \`monto\`, \`metodo\`, \`fecha\`, \`observaciones\`) VALUES\n`;
      const pVals = pagos.map(p => `(${Number(p.id)}, ${Number(p.id_cliente)}, ${escSql(p.mes)}, ${Number(p.monto || 0).toFixed(2)}, ${escSql(p.metodo)}, ${escSql(p.fecha)}, ${escSql(p.observaciones)})`);
      sql += pVals.join(',\n') + ';\n\n';
    }

    // Tabla Cortes
    sql += `-- Estructura de tabla para \`cortes\`\n`;
    sql += `CREATE TABLE IF NOT EXISTS \`cortes\` (\n`;
    sql += `  \`id\` int(11) NOT NULL AUTO_INCREMENT,\n`;
    sql += `  \`id_cliente\` int(11) NOT NULL,\n`;
    sql += `  \`motivo\` varchar(255) DEFAULT NULL,\n`;
    sql += `  \`fecha_corte\` varchar(30) DEFAULT NULL,\n`;
    sql += `  \`fecha_recon\` varchar(30) DEFAULT NULL,\n`;
    sql += `  \`estado\` varchar(30) DEFAULT 'CORTADO',\n`;
    sql += `  PRIMARY KEY (\`id\`),\n`;
    sql += `  KEY \`idx_cliente_corte\` (\`id_cliente\`)\n`;
    sql += `) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;\n\n`;

    if (cortes.length) {
      sql += `INSERT INTO \`cortes\` (\`id\`, \`id_cliente\`, \`motivo\`, \`fecha_corte\`, \`fecha_recon\`, \`estado\`) VALUES\n`;
      const cVals = cortes.map(co => `(${Number(co.id)}, ${Number(co.id_cliente)}, ${escSql(co.motivo)}, ${escSql(co.fecha_corte)}, ${escSql(co.fecha_recon)}, ${escSql(co.estado)})`);
      sql += cVals.join(',\n') + ';\n\n';
    }

    const blob = new Blob([sql], { type: 'application/sql' });
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = `leytrix_isp_mysql_${new Date().toISOString().slice(0,10)}.sql`;
    a.click();
    setTimeout(() => URL.revokeObjectURL(url), 5000);
    toast('🐬 Script MySQL (.sql) descargado exitosamente');
  } catch(err) {
    console.error('exportarMySQL:', err);
    toast('Error exportando a MySQL', 'error');
  } finally {
    if (btn) { btn.disabled = false; btn.textContent = '🐬 Descargar Script MySQL (.sql)'; }
  }
}

function abrirModal(id) {
  const m = document.getElementById(id);
  if (m) { m.classList.add('open'); m.querySelector('[id^=close],[id^=cancel]')?.focus(); }
}
function cerrarModal(id) {
  const m = document.getElementById(id);
  if (m) m.classList.remove('open');
}

document.addEventListener('keydown', e => {
  if (e.key === 'Escape') document.querySelectorAll('.modal-bg.open').forEach(m => m.classList.remove('open'));
});
document.querySelectorAll('.modal-bg').forEach(m => {
  m.addEventListener('click', e => { if (e.target === m) m.classList.remove('open'); });
});
